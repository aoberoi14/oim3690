JavaScript Demo

A read-the-code walkthrough of core JavaScript concepts.

1. Variables: var, let, const

let name = "Adi";
const year = 2026;
var score = 100;

console.log(name, year, score);

2. Functions

function greet(name) {
  return "Hello, " + name + "!";
}

const greetArrow = (name) => "Hello, " + name + "!";

console.log(greet("Adi"));
console.log(greetArrow("Babson"));

3. Arrays

let fruits = ["apple", "banana", "mango"];
fruits.push("grape");
console.log(fruits);
console.log("Length:", fruits.length);
console.log("First:", fruits[0]);

4. Objects

let person = {
  name: "Adi",
  age: 20,
  school: "Babson"
};

console.log(person.name);
console.log(person["school"]);
person.major = "Entrepreneurship";
console.log(person);

5. Conditionals

let temperature = 75;

if (temperature > 80) {
  console.log("Hot day");
} else if (temperature > 60) {
  console.log("Nice day");
} else {
  console.log("Cold day");
}

6. Loops

let numbers = [1, 2, 3, 4, 5];

numbers.forEach(function(n) {
  console.log(n * 2);
});

7. DOM Manipulation

let el = document.getElementById("demo-box");
el.textContent = "Updated by JavaScript!";
el.style.color = "green";
